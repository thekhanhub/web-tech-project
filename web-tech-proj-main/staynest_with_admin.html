Frequently Asked Questions

Quick answers to common questions

How do I cancel a booking?

Go to My Bookings and click "Cancel" on the reservation. Cancellations made 48+ hours before check-in are fully refunded.

Are the listings verified?

Yes! Every property on StayNest is verified by our team through photos, documents, and guest reviews before being listed.

What payment methods are accepted?

We accept all major credit/debit cards, JazzCash, Easypaisa, and bank transfers in supported regions.

Can I host my property on StayNest?

Absolutely! Click "Register" and choose "Become a Host". Our team will guide you through the listing process.

Is my personal data secure?

We use bank-level encryption (256-bit SSL) and never share your data with third parties without explicit consent.

How do I contact a host?

Once your booking is confirmed, you'll receive the host's contact details via email. You can also message them through our app.

db_connect.php — Database Connection 
// db_connect.php — Establishes MySQL connection for StayNest
<?php
  $host     = 'localhost';
  $dbname   = 'staynest_db';
  $username = 'root';
  $password = '';

  try {
    $pdo = new PDO("mysql:host=$host;dbname=$dbname", $username, $password);
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
  } catch (PDOException $e) {
    die("Connection failed: " . $e->getMessage());
  }

  /* CREATE TABLE IF NOT EXISTS staynest_properties (
       id          INT AUTO_INCREMENT PRIMARY KEY,
       name        VARCHAR(150)   NOT NULL,
       location    VARCHAR(150)   NOT NULL,
       type        VARCHAR(50)    NOT NULL,
       price       DECIMAL(10,2)  NOT NULL,
       rating      DECIMAL(3,2)   NOT NULL,
       amenities   VARCHAR(255),
       created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
     ); */
?>
insert.php — Insert Record (POST Handler) 
<?php
require_once 'db_connect.php';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {

  // Sanitize & validate inputs
  $name      = trim(htmlspecialchars($_POST['name']));
  $location  = trim(htmlspecialchars($_POST['location']));
  $type      = trim(htmlspecialchars($_POST['type']));
  $price     = floatval($_POST['price']);
  $rating    = floatval($_POST['rating']);
  $amenities = implode(',', $_POST['amenities'] ?? []);

  $errors = [];
  if (empty($name))               $errors[] = "Property name is required.";
  if (empty($location))           $errors[] = "Location is required.";
  if ($price <= 0)               $errors[] = "Price must be greater than 0.";
  if ($rating < 1 || $rating > 5) $errors[] = "Rating must be between 1 and 5.";

  if (empty($errors)) {
    $sql = "INSERT INTO staynest_properties (name, location, type, price, rating, amenities)
             VALUES (:name, :location, :type, :price, :rating, :amenities)";
    $stmt = $pdo->prepare($sql);
    $stmt->execute([
      ':name'      => $name,
      ':location'  => $location,
      ':type'      => $type,
      ':price'     => $price,
      ':rating'    => $rating,
      ':amenities' => $amenities
    ]);
    $_SESSION['msg'] = "✅ Property inserted successfully! ID: " . $pdo->lastInsertId();
  } else {
    $_SESSION['error'] = implode("<br>", $errors);
  }
  header("Location: admin.php");
  exit();
}
?>
update.php 
<?php
require_once 'db_connect.php';
if($_SERVER['REQUEST_METHOD']==='POST'){
  $id  = intval($_POST['id']);
  $name  = trim(htmlspecialchars($_POST['name']));
  $price = floatval($_POST['price']);
  $rating= floatval($_POST['rating']);

  $sql="UPDATE staynest_properties
    SET name=:n, price=:p, rating=:r
    WHERE id=:id";
  $stmt=$pdo->prepare($sql);
  $stmt->execute([
    ':n'=>$name,
    ':p'=>$price,
    ':r'=>$rating,
    ':id'=>$id
  ]);
  $_SESSION['msg']="✅ Record updated.";
  header("Location: admin.php");
}
?>
delete.php 
<?php
require_once 'db_connect.php';

if(isset($_GET['id'])){

  // Prevent accidental deletion

  if($id <= 0){

  }

  $sql = "DELETE FROM
    staynest_properties
    WHERE id = :id";
  $stmt = $pdo->prepare($sql);
  $stmt->execute([':id'=>$id]);

  $_SESSION['msg'] =
    "🗑️ Record #".$id." deleted.";
  header("Location: admin.php");
}
?>
